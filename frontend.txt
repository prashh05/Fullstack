//Login.jsx
import React, { useState } from 'react';
import { useNavigate } from 'react-router-dom';

const Login = () => {
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [errors, setErrors] = useState({});
  const navigate = useNavigate?.() || (()=>{});

  const handleSubmit = (e) => {
    e?.preventDefault?.();
    const errs = {};
    if (!email) errs.email = 'Email is required';
    if (!password) errs.password = 'Password is required';
    setErrors(errs);
    if (Object.keys(errs).length === 0) {
      // pretend login
      navigate('/');
    }
  };

  return (
    <div>
      <h1>Login</h1>
      <form onSubmit={handleSubmit}>
        <div>
          <label>Email</label>
          <input aria-label="Email" value={email} onChange={e=>setEmail(e.target.value)} />
          {errors.email && <div>{errors.email}</div>}
        </div>
        <div>
          <label>Password</label>
          <input aria-label="Password" type="password" value={password} onChange={e=>setPassword(e.target.value)} />
          {errors.password && <div>{errors.password}</div>}
        </div>
        <button type="button" onClick={handleSubmit}>Login</button>
      </form>
    </div>
  );
};

export default Login;


//Register.jsx
import React, { useState } from 'react';

const Register = () => {
  const [values, setValues] = useState({});
  const [errors, setErrors] = useState({});
  const onSubmit = (e) => {
    e?.preventDefault?.();
    const errs = {};
    if (!values.firstName) errs.firstName = 'First Name is required';
    if (!values.lastName) errs.lastName = 'Last Name is required';
    if (!values.email) errs.email = 'Email is required';
    if (!values.mobile) errs.mobile = 'Mobile Number is required';
    if (!values.password) errs.password = 'Password is required';
    if (!values.confirm) errs.confirm = 'Confirm Password is required';
    setErrors(errs);
  };

  return (
    <div>
      <h2>Create Your Account</h2>
      <form onSubmit={onSubmit}>
        <input placeholder="First Name" onChange={e=>setValues({...values, firstName: e.target.value})} />
        {errors.firstName && <div>First Name is required</div>}
        <input placeholder="Last Name" onChange={e=>setValues({...values, lastName: e.target.value})} />
        {errors.lastName && <div>Last Name is required</div>}
        <input placeholder="Email" onChange={e=>setValues({...values, email: e.target.value})} />
        {errors.email && <div>Email is required</div>}
        <input placeholder="Mobile Number" onChange={e=>setValues({...values, mobile: e.target.value})} />
        {errors.mobile && <div>Mobile Number is required</div>}
        <input placeholder="Password" type="password" onChange={e=>setValues({...values, password: e.target.value})} />
        {errors.password && <div>Password is required</div>}
        <input placeholder="Confirm Password" type="password" onChange={e=>setValues({...values, confirm: e.target.value})} />
        {errors.confirm && <div>Confirm Password is required</div>}
        <button type="button" onClick={onSubmit}>Register</button>
      </form>
    </div>
  );
};

export default Register;

//CreateTask.jsx
import React, { useState } from 'react';

const CreateTask = () => {
  // editId read from localStorage by tests
  const editId = (typeof window !== 'undefined' && window.localStorage?.getItem('editId')) || '';
  const [values, setValues] = useState({ taskName: '', description: '', assignee: '', dueDate: '' });
  const [errors, setErrors] = useState({});

  const onSubmit = (e) => {
    e?.preventDefault?.();
    const errs = {};
    if (!values.taskName) errs.taskName = 'Task name is required';
    if (!values.description) errs.description = 'Description is required';
    if (!values.assignee) errs.assignee = 'Assignee is required';
    if (!values.dueDate) errs.dueDate = 'Due date is required';
    setErrors(errs);
  };

  return (
    <div>
      <h2>Add New Task</h2>
      <form onSubmit={onSubmit}>
        <input placeholder="Task Name" value={values.taskName} onChange={e=>setValues({...values, taskName: e.target.value})} />
        {errors.taskName && <div>{errors.taskName}</div>}
        <input placeholder="Description" value={values.description} onChange={e=>setValues({...values, description: e.target.value})} />
        {errors.description && <div>{errors.description}</div>}
        <input placeholder="Assignee" value={values.assignee} onChange={e=>setValues({...values, assignee: e.target.value})} />
        {errors.assignee && <div>{errors.assignee}</div>}
        <input placeholder="Due Date" type="date" value={values.dueDate} onChange={e=>setValues({...values, dueDate: e.target.value})} />
        {errors.dueDate && <div>{errors.dueDate}</div>}
        <button type="button" onClick={onSubmit}>{editId ? 'Update Task' : 'Add Task'}</button>
      </form>
    </div>
  );
};

export default CreateTask;
